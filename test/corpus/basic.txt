==================
Call-scoped references
==================

fn sum(&int left, &int right) -> int {
    return left + right;
}

fn increment(&mut int value) -> int {
    value += 1;
    return value;
}

fn main() -> int {
    let mut int count = 40;
    print(sum(&count, &count));
    increment(&mut count);
    return count;
}
